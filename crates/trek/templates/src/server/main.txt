-- Server Side Script: {{ resource_name }}

CreateThread(function()
    print("^2[{{ resource_name }}]^7 Server script initialized.")
end)

RegisterServerEvent('{{ resource_name }}:server:callback')
AddEventHandler('{{ resource_name }}:server:callback', function(cb)
    local src = source
    if Utils.FrameworkName == 'qbcore' and Utils.Framework then
        local Player = Utils.Framework.Functions.GetPlayer(src)
        if Player and cb then cb(Player.PlayerData.money) end
    elseif Utils.FrameworkName == 'esx' and Utils.Framework then
        local xPlayer = Utils.Framework.GetPlayerFromId(src)
        if xPlayer and cb then cb(xPlayer.getAccounts()) end
    elseif Utils.FrameworkName == 'qbox' and Utils.Framework then
        local Player = exports.qbx_core:GetPlayer(src) or (Utils.Framework.Functions and Utils.Framework.Functions.GetPlayer(src))
        if Player and cb then cb(Player.PlayerData.money) end
    end
end)
